public class TimerQuest : MonoBehaviour
{
    [SerializeField] private TextMeshProUGUI _timerText;
    
    private float _timerDuration = 600f; // 10 минут
    private float _currentTime; // Переименовано для ясности
    
    public float Timer => _currentTime;
    public event Action OnTimerExpired;

    public void ActivateTimer()
    {
        _currentTime = _timerDuration; // Сброс времени
        StartCoroutine(UpdateTimer());
    }

    private IEnumerator UpdateTimer()
    {
        while (_currentTime > 0)
        {
            // Уменьшаем время КАЖДЫЙ КАДР без задержки
            _currentTime -= Time.deltaTime;
            UpdateTimerText(_currentTime);
            
            // Важно: yield без задержки!
            yield return null;
        }
        
        _currentTime = 0;
        UpdateTimerText(_currentTime);
        OnTimerExpired?.Invoke();
    }

    private void UpdateTimerText(float time)
    {
        time = Mathf.Max(0, time); // Защита от отрицательных значений
        int minutes = Mathf.FloorToInt(time / 60);
        int seconds = Mathf.FloorToInt(time % 60);
        
        // Исправлено форматирование (убраны пробелы в скобках)
        _timerText.text = string.Format("{0:00}:{1:00}", minutes, seconds);
    }
}
